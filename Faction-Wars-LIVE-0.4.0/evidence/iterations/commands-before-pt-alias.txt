import {RULES,ordered} from '../rules/rules.js';
import {birth,living,effect,diagnostic} from '../core/state.js';
const aliases={};for(const [value,names] of Object.entries({R:['red','rubra'],S:['blue','safira'],G:['guerreiro','warrior','g','w'],D:['defensor','guardian','d','gd'],A:['arqueiro','archer','a'],left:['esquerda','left'],mid:['centro','mid'],right:['direita','right'],offensive:['atacar','attack'],defensive:['defender','defend'],balanced:['equilibrar','balance','balanced'],help:['ajuda','help'],status:['status','meustatus','mystatus']}))for(const n of names)aliases['!'+n]=value;
export function tokens(text){return String(text).trim().split(/\s+/).map(x=>aliases[x.toLowerCase()]).filter(Boolean);}
export function join(s,id,f,name=id){const p=s.people[id];if(p){if(p.faction!==f)diagnostic(s,'faction-fixed',id);return;}if(s.phase!=='preparation'){if(!s.nextQueue[id])s.nextQueue[id]={id,faction:f,name:String(name).slice(0,80)};diagnostic(s,'queued-next-match',id);return;}s.people[id]={id,faction:f,name:String(name).slice(0,80),E:0,V:0,likes:0,rest:0,last:s.phase==='preparation'?s.phaseMs-25000:s.combatMs,represented:0,statusAt:-30000};effect(s,'join',{actorId:id,faction:f});}
export function command(s,id,text,name=id){let count=0;const parsed=tokens(text);if(!parsed.length)diagnostic(s,'unknown-command',id);for(const token of parsed){if(count>=3){diagnostic(s,'three-action-limit',id);break;}if(['R','S'].includes(token)){join(s,id,token,name);count++;continue;}const p=s.people[id];if(token==='help'||token==='status'){if(p&&s.combatMs-p.statusAt>=30000){p.statusAt=s.combatMs;effect(s,'local-status',{actorId:id});}count++;continue;}if(!p||s.phase!=='combat')continue;const t=s.teams[p.faction];if(['A','G','D'].includes(token)){if(!t.window)t.window={id:++t.windowSeq,start:s.combatMs,end:s.combatMs+10000,votes:{}};const old=t.window.votes[id];if(old?.kind!==token)t.window.votes[id]={kind:token,lane:null};count++;p.last=s.combatMs;}else if(['left','mid','right'].includes(token)){const lane=['left','mid','right'].indexOf(token);if(t.window?.votes[id])t.window.votes[id].lane=lane;else t.strategic.lane[id]=lane;count++;p.last=s.combatMs;}else if(RULES.stances[token]){t.strategic.stance[id]=token;count++;p.last=s.combatMs;}}}
export function rotate(options,cycle,team,key){if(options.length===1)return options[0];for(let i=0;i<cycle.length;i++){const index=(team[key]+i)%cycle.length;if(options.includes(cycle[index])){team[key]=(index+1)%cycle.length;return cycle[index];}}}
export function representative(s,f){const t=s.teams[f],live=living(s);const ps=Object.values(s.people).filter(p=>p.faction===f&&!live.some(u=>u.actorId===p.id&&u.champion));ps.sort((a,b)=>Number(a.represented>0)-Number(b.represented>0)||(5*b.E+10*b.V)-(5*a.E+10*a.V)||a.id.localeCompare(b.id));if(!ps.length)return null;const best=ps[0],ties=ps.filter(p=>(p.represented>0)===(best.represented>0)&&5*p.E+10*p.V===5*best.E+10*best.V).sort((a,b)=>a.id.localeCompare(b.id));const p=ties.find(p=>p.id>t.repTurn)||ties[0];t.repTurn=p.id;return p;}
export function buy(s,f,kind,requested,id){const t=s.teams[f],all=living(s).filter(u=>u.kind!=='C');if(all.length>=32||all.filter(u=>u.faction===f).length>=16){effect(s,'purchase-failed',{purchaseId:id,reason:'cap'});return null;}const count=l=>all.filter(u=>u.faction===f&&u.lane===l).length;let lane=requested;if(lane===null||count(lane)>=8){const min=Math.min(...[0,1,2].map(count));if(min>=8)return null;lane=rotate([0,1,2].filter(l=>count(l)===min),[0,1,2],t,'laneTurn');}const has=all.some(u=>u.faction===f&&u.lane===lane&&u.kind===kind&&u.champion);const champion=!has&&t.credits>=RULES.costs[kind][1],cost=RULES.costs[kind][champion?1:0];if(t.credits<cost){effect(s,'purchase-failed',{purchaseId:id,reason:'credits'});return null;}const p=champion?representative(s,f):null;t.credits-=cost;t.decay=null;if(p)p.represented++;const u=birth(s,f,lane,kind,champion,p?.id??null,`${id}:unit`);effect(s,'purchase',{purchaseId:id,unitId:u.id,cost,actorId:p?.id??null});return u;}
export function closeWindow(s,f,purchase=true){const t=s.teams[f],w=t.window;if(!w)return;const votes=Object.entries(w.votes).sort(([a],[b])=>a.localeCompare(b));const counts={A:0,G:0,D:0};for(const [id,v] of votes){counts[v.kind]++;s.people[id].V+=v.lane===null?1:2;}const ranked=[];while(ranked.length<2){const candidates=['A','G','D'].filter(k=>counts[k]>0&&!ranked.includes(k));if(!candidates.length)break;const max=Math.max(...candidates.map(k=>counts[k]));ranked.push(rotate(candidates.filter(k=>counts[k]===max),['A','G','D'],t,'classTurn'));}if(ranked.length===1)ranked.push(ranked[0]);if(purchase)ranked.forEach((kind,i)=>{const ls=[0,0,0];for(const [,v] of votes)if(v.kind===kind&&v.lane!==null)ls[v.lane]++;const max=Math.max(...ls),opts=[0,1,2].filter(l=>ls[l]===max);const live=living(s).filter(u=>u.faction===f&&u.kind!=='C');const n=l=>live.filter(u=>u.lane===l).length;const min=Math.min(...opts.map(n));const lane=rotate(opts.filter(l=>n(l)===min),[0,1,2],t,'laneTurn');buy(s,f,kind,lane,`${s.id}:purchase:${f}:${w.id}:${i+1}`);});t.window=null;}
export function closeStrategic(s,f){const t=s.teams[f];for(const category of ['stance','lane']){const counts={};for(const [id,v] of Object.entries(t.strategic[category])){counts[v]=(counts[v]??0)+1;s.people[id].V++;}const keys=Object.keys(counts),max=Math.max(0,...Object.values(counts)),wins=keys.filter(k=>counts[k]===max);if(category==='stance'){if(wins.length===1)t.stance=wins[0];}else if(wins.length===1)t.strategy=Number(wins[0]);}t.strategic={stance:{},lane:{}};t.nextStrategy+=10000;}
